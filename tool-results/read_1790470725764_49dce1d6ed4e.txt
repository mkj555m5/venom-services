   297→# ── PROTOBUF helpers ──
   298→def _varint(n):
   299→    out = []
   300→    while True:
   301→        b = n & 0x7F; n >>= 7
   302→        if n: b |= 0x80
   303→        out.append(b)
   304→        if not n: break
   305→    return bytes(out)
   306→
   307→def _field(fn, v):
   308→    if isinstance(v, bool):  return _varint((fn<<3)|0) + _varint(int(v))
   309→    if isinstance(v, int):   return _varint((fn<<3)|0) + _varint(v)
   310→    if isinstance(v, bytes): return _varint((fn<<3)|2) + _varint(len(v)) + v
   311→    if isinstance(v, str):
   312→        b = v.encode()
   313→        return _varint((fn<<3)|2) + _varint(len(b)) + b
   314→    raise TypeError(f"unsupported {type(v)}")
   315→
   316→def assemble_proto(fields: dict) -> bytes:
   317→    return b''.join(_field(int(k), v) for k, v in fields.items())
   318→
   319→def _read_varint(data, pos):
   320→    result = 0; shift = 0
   321→    while pos < len(data):
   322→        bv = data[pos]; pos += 1
   323→        result |= (bv & 0x7F) << shift
   324→        if not (bv & 0x80): break
   325→        shift += 7
   326→    return result, pos
   327→
   328→def _try_parse_proto(raw: bytes):
   329→    if not raw: return None
   330→    out = {}; pos = 0; seen = 0
   331→    try:
   332→        while pos < len(raw):
   333→            tag, pos = _read_varint(raw, pos)
   334→            if tag == 0: return None
   335→            fn, wt = tag >> 3, tag & 7
   336→            if fn == 0 or fn > 536870911: return None
   337→            if wt == 0:
   338→                v, pos = _read_varint(raw, pos)
   339→                out[fn] = v
   340→            elif wt == 2:
   341→                ln, pos = _read_varint(raw, pos)
   342→                if pos + ln > len(raw): return None
   343→                val = raw[pos:pos+ln]; pos += ln
   344→                n = _try_parse_proto(val)
   345→                if n and len(n) >= 1:
   346→                    out[fn] = n
   347→                else:
   348→                    try:    out[fn] = val.decode('utf-8')
   349→                    except: out[fn] = val.hex()
   350→            elif wt == 5:
   351→                if pos+4 > len(raw): return None
   352→                out[fn] = raw[pos:pos+4].hex(); pos += 4
   353→            elif wt == 1:
   354→                if pos+8 > len(raw): return None
   355→                out[fn] = raw[pos:pos+8].hex(); pos += 8
   356→            else: return None
   357→            seen += 1
   358→        return out if seen else None
   359→    except: return None
